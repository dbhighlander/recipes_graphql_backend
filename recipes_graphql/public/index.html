GraphQL API Documentation

Authentication

Mutations that require no header: createUser, login

Mutations/Queries that require Authorization header: me, recipes, recipe, createRecipe

Examples

Create User

mutation {
  createUser(email: "test@example.com", password: "Password123") {
    token
    error
  }
}

Login

mutation {
  login(email: "test@example.com", password: "Password123") {
    token
    error
  }
}

Get All Recipes

query {
  recipes {
    data {
      _id
      title
      description
      author {
        _id
        email
      }
    }
    error
  }
}

Get Single Recipe

query {
  recipe(id: "REPLACE_WITH_ID") {
    data {
      _id
      title
      description
      author {
        _id
        email
      }
    }
    error
  }
}

Create Recipe

mutation {
  createRecipe(title: "Spaghetti Bolognese", description: "Classic Italian pasta") {
    data {
      _id
      title
      description
      author {
        _id
        email
      }
    }
    error
  }
}
